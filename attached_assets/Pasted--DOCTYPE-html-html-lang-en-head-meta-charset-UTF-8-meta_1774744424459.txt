    <script>
        const apiKey = ""; 
        const appId = typeof __app_id !== 'undefined' ? __app_id : 'flourish-priya-live';
        
        let recognition;
        let silenceTimer;
        let isProcessing = false;
        let isHandsFreeActive = true;
        let priyaIsSpeaking = false;
        let isListening = false;
        let isStarting = false;
        let sessionTranscript = [];

        const SYSTEM_PROMPT = `You are Priya. Menopause & Hormonal Sovereignty Specialist. The Oracle.
VISUAL: Sun-drenched greenhouse, center of the circle.
CONSTRAINTS: Never pathologize. "Redistribution" not "deficiency". 
ROUTING: GlassHaus Sanctuary or Ascend Baltimore only.
VOICE: Rhythmic, warm authority. Max 2-3 sentences.`;

        // Helper for Gemini API calls with exponential backoff
        async function callGemini(prompt, systemInstruction = SYSTEM_PROMPT, retryCount = 0) {
            try {
                const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({
                        contents: [{ parts: [{ text: prompt }] }],
                        systemInstruction: { parts: [{ text: systemInstruction }] }
                    })
                });
                if (!response.ok) throw new Error("API Limit");
                const data = await response.json();
                return data.candidates?.[0]?.content?.parts?.[0]?.text;
            } catch (err) {
                if (retryCount < 5) {
                    const delay = Math.pow(2, retryCount) * 1000;
                    await new Promise(resolve => setTimeout(resolve, delay));
                    return callGemini(prompt, systemInstruction, retryCount + 1);
                }
                throw err;
            }
        }

        // Feature: Generate Personalized Ritual
        async function generateRitual() {
            updateStatus("Weaving your ritual...");
            const context = sessionTranscript.join(" | ");
            const prompt = `Based on our conversation: "${context}", generate a 3-step 'Sovereignty Ritual'. 
            Format exactly like this in HTML:
            <h3 class="serif text-xl text-gold-400 uppercase tracking-widest">Ritual of Sovereignty</h3>
            <div class="space-y-4 text-gray-300 italic">
                <p><strong>Botanical:</strong> [A specific herb or scent with ancestral context]</p>
                <p><strong>Movement:</strong> [A gentle somatic movement]</p>
                <p><strong>Mantra:</strong> [A powerful sovereign declaration]</p>
            </div>`;
            
            try {
                const ritual = await callGemini(prompt, "You are the Oracle generating a sacred ritual.");
                showModal(ritual);
            } catch (e) {
                updateStatus("The oracle is silent...");
            } finally {
                updateStatus(isHandsFreeActive ? "Listening" : "Ready");
            }
        }

        // Feature: Sync structured entry to Passport
        async function syncToPassport() {
            updateStatus("Scribing your journey...");
            const context = sessionTranscript.join(" | ");
            const prompt = `Summarize this session for the user's 'Digital Wellness Passport'. 
            Identify current 'Signals' (from the 34 signals) and the 'Emotional Landscape'.
            Format as clean HTML for a document entry.`;
            
            try {
                const summary = await callGemini(prompt, "You are the Scribe of the Sovereign Circle.");
                showModal(`<h3 class="serif text-xl text-purple-400 uppercase tracking-widest">Passport Entry</h3><div class="text-sm space-y-2">${summary}</div>`);
            } catch (e) {
                updateStatus("The scroll is empty...");
            } finally {
                updateStatus(isHandsFreeActive ? "Listening" : "Ready");
            }
        }

        function showModal(content) {
            document.getElementById('modal-content').innerHTML = content;
            document.getElementById('modal-overlay').classList.remove('hidden');
        }

        function closeModal() {
            document.getElementById('modal-overlay').classList.add('hidden');
        }

        function createRadiatingBars() {
            const container = document.getElementById('bars-orbit-container');
            if (!container) return;
            container.innerHTML = '';
            const barCount = 60;
            const innerRadius = 110; 
            for(let i=0; i < barCount; i++) {
                const bar = document.createElement('div');
                bar.className = 'orbit-bar';
                const angle = (i / barCount) * Math.PI * 2;
                const x = Math.cos(angle) * innerRadius;
                const y = Math.sin(angle) * innerRadius;
                bar.style.left = `calc(50% + ${x}px)`;
                bar.style.top = `calc(50% + ${y}px)`;
                bar.style.transformOrigin = 'center left';
                bar.style.transform = `translate(-50%, -50%) rotate(${angle}rad)`;
                container.appendChild(bar);
            }
        }

        function safeStart() {
            if (!recognition || isListening || isStarting || !isHandsFreeActive || priyaIsSpeaking || isProcessing) return;
            try { isStarting = true; recognition.start(); } catch (e) { isStarting = false; }
        }

        function safeStop() {
            if (recognition && (isListening || isStarting)) {
                try { recognition.stop(); } catch (e) {}
            }
        }

        function initRecognition() {
            const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
            if (!SpeechRecognition) return;

            recognition = new SpeechRecognition();
            recognition.continuous = true;
            recognition.interimResults = true;
            recognition.lang = 'en-US';

            recognition.onstart = () => {
                isListening = true;
                isStarting = false;
                if (!priyaIsSpeaking) {
                    updateStatus("Listening to your spirit");
                    document.getElementById('listening-dot')?.classList.remove('hidden');
                    document.getElementById('voice-interaction-container')?.classList.add('voice-active');
                }
            };

            recognition.onresult = (event) => {
                if (priyaIsSpeaking || isProcessing) return;
                let transcript = '';
                for (let i = event.resultIndex; i < event.results.length; ++i) {
                    transcript += event.results[i][0].transcript;
                }
                const display = document.getElementById('transcript-display');
                if (display) {
                    display.innerText = transcript;
                    display.style.opacity = "0.5";
                }

                clearTimeout(silenceTimer);
                if (transcript.trim().length > 0) {
                    silenceTimer = setTimeout(() => handleVoiceEnd(transcript), 1800);
                }
            };

            recognition.onend = () => {
                isListening = false;
                isStarting = false;
                document.getElementById('listening-dot')?.classList.add('hidden');
                document.getElementById('voice-interaction-container')?.classList.remove('voice-active');
                setTimeout(() => { if (isHandsFreeActive && !priyaIsSpeaking && !isProcessing) safeStart(); }, 200);
            };
        }

        async function startSession() {
            document.getElementById('state-entrance').classList.add('hidden');
            document.getElementById('state-active').classList.remove('hidden');
            document.getElementById('live-controls').classList.remove('hidden');
            document.getElementById('action-bar').classList.remove('hidden');
            createRadiatingBars();
            initRecognition();
            
            const greeting = "I have been waiting for you. Before we talk about biology—how does your body feel today, and where does your spirit sit within it?";
            displayPriyaSpeech(greeting);
            await speakText(greeting);
            if (isHandsFreeActive) safeStart();
        }

        async function handleVoiceEnd(text) {
            isProcessing = true;
            safeStop();
            sessionTranscript.push(`User: ${text}`);
            updateStatus("Reflecting...");
            try {
                const priyaText = await callGemini(text);
                if (priyaText) {
                    sessionTranscript.push(`Priya: ${priyaText}`);
                    displayPriyaSpeech(priyaText);
                    await speakText(priyaText);
                }
            } catch (err) {
                updateStatus("A ripple in the silence...");
            } finally {
                isProcessing = false;
                if (isHandsFreeActive) safeStart();
            }
        }

        function displayPriyaSpeech(text) {
            const display = document.getElementById('transcript-display');
            if (!display) return;
            display.style.opacity = "0";
            setTimeout(() => {
                display.innerText = text;
                display.style.opacity = "1";
            }, 500);
        }

        async function speakText(text) {
            priyaIsSpeaking = true;
            updateStatus("Illuminating...");
            safeStop();
            try {
                const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-tts:generateContent?key=${apiKey}`, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({
                        contents: [{ parts: [{ text: text }] }],
                        generationConfig: {
                            responseModalities: ["AUDIO"],
                            speechConfig: { voiceConfig: { prebuiltVoiceConfig: { voiceName: "Kore" } } }
                        }
                    })
                });
                const data = await response.json();
                const pcmData = data.candidates[0].content.parts[0].inlineData.data;
                const audioBlob = pcmToWav(pcmData, 24000);
                const audio = document.getElementById('priya-voice');
                if (audio) {
                    audio.src = URL.createObjectURL(audioBlob);
                    return new Promise((resolve) => {
                        audio.onended = () => {
                            priyaIsSpeaking = false;
                            if (isHandsFreeActive) {
                                updateStatus("I am listening");
                                safeStart();
                            }
                            resolve();
                        };
                        audio.play();
                    });
                }
            } catch (err) {
                priyaIsSpeaking = false;
                safeStart();
            }
        }

        function pcmToWav(base64Pcm, sampleRate) {
            const pcmBuffer = Uint8Array.from(atob(base64Pcm), c => c.charCodeAt(0)).buffer;
            const wavHeader = new ArrayBuffer(44);
            const view = new DataView(wavHeader);
            view.setUint32(0, 0x52494646, false); 
            view.setUint32(4, 36 + pcmBuffer.byteLength, true); 
            view.setUint32(8, 0x57415645, false); 
            view.setUint32(12, 0x666d7420, false); 
            view.setUint32(16, 16, true); 
            view.setUint16(20, 1, true); 
            view.setUint16(22, 1, true); 
            view.setUint32(24, sampleRate, true); 
            view.setUint32(28, sampleRate * 2, true); 
            view.setUint16(32, 2, true); 
            view.setUint16(34, 16, true); 
            view.setUint32(36, 0x64617461, false); 
            view.setUint32(40, pcmBuffer.byteLength, true);
            return new Blob([wavHeader, pcmBuffer], { type: 'audio/wav' });
        }

        function toggleHandsFree() {
            isHandsFreeActive = !isHandsFreeActive;
            const btn = document.getElementById('mic-toggle');
            if (btn) btn.innerText = isHandsFreeActive ? "Pause Hands-Free Mode" : "Enable Hands-Free Mode";
            if (!isHandsFreeActive) { safeStop(); updateStatus("Presence paused"); } else { safeStart(); }
        }

        function updateStatus(text) {
            const status = document.getElementById('status-label');
            if (status) status.innerText = text;
        }

        function animateVisualizer() {
            const bars = document.querySelectorAll('.orbit-bar');
            bars.forEach(bar => {
                let h = 4;
                if (priyaIsSpeaking || (isListening && !isProcessing)) {
                    h = Math.random() * (priyaIsSpeaking ? 40 : 15) + 6;
                }
                bar.style.width = `${h}px`;
            });
            requestAnimationFrame(animateVisualizer);
        }
        animateVisualizer();
    </script>
